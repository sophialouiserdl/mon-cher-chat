    function dateOffset(days) {
      const d = new Date();
      d.setHours(12, 0, 0, 0);
      d.setDate(d.getDate() + days);
      return d.toISOString().slice(0, 10);
    }

    function todayISO() {

    function fmtDate(iso) {

    function daysBetween(fromISO, to = new Date()) {
      const from = new Date(`${fromISO}T12:00:00`);
      const until = new Date(to);
      until.setHours(12, 0, 0, 0);
      return Math.floor((until - from) / DAY);
    }

    function weakSessionHash(value) {
      let hash = 0;
      const input = String(value || "");
      for (let i = 0; i < input.length; i++) {
        hash = ((hash << 5) - hash) + input.charCodeAt(i);
        hash |= 0;
      }
      return `session:${Math.abs(hash)}`;
    }

          .sort((a, b) => a.bpo.localeCompare(b.bpo) || a.id - b.id)
          .forEach(c => {

            const selected = auditors.find(a => counts[a.id] === min) || auditors[0];
            c.auditorId = selected.id;
            c.assignedAt = c.assignedAt || todayISO();
            counts[selected.id]++;

        } else {
          this.store.audits.push({ id: nextId(this.store.audits), ...payload });
        }
        return { case: { ...c }, audit: payload };
      }

    function getCriteria(type) {

    function answerWeight(answer) {
      if (answer === ANSWERS.OK) return 1;
      if (answer === ANSWERS.PARTIAL) return 0.5;
      return 0;
    }

    function calculateScore(criteria) {
      const total = criteria.reduce((sum, item) => sum + Number(item.peso || 0), 0) || 100;
      const score = criteria.reduce((sum, item) => sum + Number(item.peso || 0) * answerWeight(item.resposta), 0);
      return Math.round((score / total) * 1000) / 10;
    }

    function classify(score) {
      if (score >= 90) return "Excelente";
      if (score >= 70) return "Regular";
      return "Crítico";
    }

    function classVariant(classificacao) {
      if (classificacao === "Excelente") return "success";

    }

    function statusBadge(status) {
      const map = {
        [STATUS.CONCLUIDO]: ["success", "ph-check-circle", "Concluído"],
        [STATUS.EM_AUDITORIA]: ["info", "ph-pencil-simple-line", "Em auditoria"],
        [STATUS.BLOQUEADO]: ["neutral", "ph-lock", "Bloqueado"],
        [STATUS.ERRO_SYNC]: ["danger", "ph-warning-octagon", "Erro de sincronização"],
        [STATUS.PENDENTE]: ["warning", "ph-clock", "Pendente"]
      };
      const [variant, iconName, label] = map[status] || map[STATUS.PENDENTE];
      return `<span class="badge ${variant}">${icon(iconName)} ${label}</span>`;
    }

      }
      const daysUsed = Math.max(0, daysBetween(c.assignedAt));
      const daysLeft = state.config.slaDays - daysUsed;
      const overdue = c.status !== STATUS.CONCLUIDO && daysLeft < 0;
      const percent = Math.min(100, Math.max(0, (daysUsed / state.config.slaDays) * 100));
      let variant = "success";
      let color = "var(--success)";
      if (overdue || daysLeft <= 3) {
        variant = "danger";
        color = "var(--danger)";
      } else if (daysLeft <= 7) {
        variant = "warning";
        color = "var(--warning)";
      }
      const label = c.status === STATUS.CONCLUIDO
        ? "SLA encerrado"
        : overdue
          ? `${Math.abs(daysLeft)} dia(s) em atraso`
          : `${daysLeft} dia(s) restantes`;
      return { daysUsed, daysLeft, overdue, percent, label, variant, color };
    }

    function handleLogin(event) {
      event.preventDefault();
      if (state.loginLoading) return;
      const login = document.getElementById("loginUser").value.trim();
      const senha = document.getElementById("loginPass").value;
      clearLoginErrors();
      let hasError = false;
      if (!login) {
        markLoginError("loginUser", "Informe o usuário.");
        hasError = true;
      }
      if (!senha) {
        markLoginError("loginPass", "Informe a senha.");
        hasError = true;
      }
      if (hasError) return;

      state.loginLoading = true;
      const btn = document.getElementById("loginButton");
      btn.innerHTML = `<span class="spinner"></span> Entrando`;
      btn.disabled = true;

      setTimeout(() => {
        const user = DataService.login(login, senha);
        state.loginLoading = false;
        btn.disabled = false;
        btn.innerHTML = `${icon("ph-sign-in")} Entrar`;
        if (!user) {
          document.getElementById("loginError").textContent = "Usuário ou senha inválidos, ou usuário inativo.";
          return;
        }
        state.currentUser = user;
        state.currentTab = hasManagerAccess(user) ? "adminDashboard" : "queue";
        state.currentAudit = null;
        document.getElementById("loginView").classList.add("hidden");
        document.getElementById("appView").classList.remove("hidden");
        renderApp();
        showToast(`Bem-vindo, ${user.nome}.`, "success");
      }, 650);
    }

    function handleInitialSetup(event) {
      event.preventDefault();
      const nome = document.getElementById("setupName").value.trim();
      const login = document.getElementById("setupLogin").value.trim();
      const senha = document.getElementById("setupPass").value;
      clearSetupErrors();
      let hasError = false;
      if (!nome) { markSetupError("setupName", "Informe o nome do admin."); hasError = true; }
      if (!login) { markSetupError("setupLogin", "Informe um login."); hasError = true; }
      if (!senha || senha.length < 6) { markSetupError("setupPass", "Use uma senha com pelo menos 6 caracteres."); hasError = true; }

      try {

        state.currentTab = "adminDashboard";

    function startAudit(caseId, context) {
      let c;
      try {
        c = DataService.lockCase(caseId, state.currentUser.id);
        refreshStateFromService();
      } catch (err) {
        showToast(err.message || "Não foi possível reservar este caso.", "error");
        return;
      }
      const criteria = getCriteria(c.tipoAuditoria).map(crit => ({
        criterioNome: crit.nome,
        peso: crit.peso,
        resposta: "",
        observacaoIndividual: ""
      }));
      state.currentAudit = {
        mode: "new",
        context: context === "admin" ? "admin" : "auditor",
        step: 1,
        caseId: c.id,
        auditorId: state.currentUser.id,
        editingAuditId: null,
        caseData: { ...c, cliente: c.cliente || (c.tipoAuditoria === "Escrita Mensageria" ? c.atendente : "") },
        criteria,
        openCriteria: {},
        manualCreated: false,
        manualSimplified: false,
        manualScore: "",
        observacaoGeral: "",
        errors: {},
        lastSavedAt: null,
        dirty: false
      };
      state.saveStatus = DataService.remoteConnected ? "salvo" : "aguardando banco de dados";
      state.currentTab = context === "admin" ? "createAudit" : "newAudit";
      renderApp();
    }

    function blankCaseData(type = "Telefonica Basica") {
      return {
        id: null,
        auditorId: state.currentUser.id,
        auditorNome: state.currentUser.nome,
        atendente: "",
        cliente: "",
        bpo: "",
        tipoAuditoria: type,
        tipoLigacao: type === "Escrita Mensageria" ? "" : "Receptiva",
        tempoLigacao: "",
        propostaConsulta: "",
        contrato: "",
        idCaso: generateCaseCode(),
        status: STATUS.PENDENTE,
        criadoEm: todayISO(),
        importadoEm: todayISO(),
        assignedAt: todayISO(),
        completedAt: null,
        createdManually: true
      };
    }

    function startManualAudit(context) {
      const type = "Telefonica Basica";
      state.currentAudit = {
        mode: "new",
        context: context === "admin" ? "admin" : "auditor",
        step: 1,
        caseId: null,
        auditorId: state.currentUser.id,
        editingAuditId: null,
        caseData: blankCaseData(type),
        criteria: getCriteria(type).map(crit => ({
          criterioNome: crit.nome,
          peso: crit.peso,
          resposta: "",
          observacaoIndividual: ""
        })),
        openCriteria: {},
        manualCreated: true,
        manualSimplified: false,
        manualScore: "",
        observacaoGeral: "",
        errors: {},
        lastSavedAt: null,
        dirty: false
      };
      state.saveStatus = DataService.remoteConnected ? "salvo" : "aguardando banco de dados";
      state.currentTab = context === "admin" ? "createAudit" : "newAudit";
      renderApp();
    }

    function updateAuditType(type) {
      if (!state.currentAudit || !state.currentAudit.manualCreated) return;
      const previous = state.currentAudit.caseData;
      state.currentAudit.caseData = {
        ...blankCaseData(type),
        idCaso: previous.idCaso,
        bpo: previous.bpo || "",
        atendente: type === "Escrita Mensageria" ? "" : (previous.atendente || ""),
        cliente: type === "Escrita Mensageria" ? (previous.cliente || previous.atendente || "") : "",
        propostaConsulta: type === "Escrita Mensageria" ? (previous.propostaConsulta || "") : "",
        contrato: type === "Escrita Mensageria" ? (previous.contrato || "") : "",
        tipoLigacao: type === "Escrita Mensageria" ? "" : (previous.tipoLigacao || "Receptiva")
      };
      state.currentAudit.criteria = getCriteria(type).map(crit => ({
        criterioNome: crit.nome,
        peso: crit.peso,
        resposta: "",
        observacaoIndividual: ""
      }));
      state.currentAudit.manualSimplified = false;
      state.currentAudit.manualScore = "";
      state.currentAudit.errors = {};
      markAuditDirty();
      autosaveNow(false);
      renderActiveTab();
    }

    function openAuditReadByCase(caseId) {
      const audit = auditForCase(caseId);
      if (!audit) return showToast("Ainda não existe auditoria salva para este caso.", "warning");
      openAudit(audit.id, "read");
    }

    function openAudit(auditId, mode = "read") {
      const a = auditById(auditId);
      if (!a) return showToast("Auditoria não encontrada.", "error");
      const c = caseById(a.caseId);
      state.currentAudit = {
        mode,
        context: hasManagerAccess() ? "admin" : "auditor",
        step: 3,
        caseId: c.id,
        auditorId: a.auditorId,
        editingAuditId: a.id,
        caseData: { ...c },
        criteria: a.criterios.map(item => ({ ...item })),
        openCriteria: {},
        manualSimplified: !!a.manualSimplified,
        manualScore: a.manualSimplified ? String(a.notaFinal) : "",
        observacaoGeral: a.observacaoGeral || "",
        errors: {},
        lastSavedAt: null,
        dirty: false
      };
      state.saveStatus = DataService.remoteConnected ? "salvo" : "aguardando banco de dados";
      state.currentTab = hasManagerAccess() ? "createAudit" : "newAudit";
      renderApp();
    }

    function editCurrentAudit() {
      if (!state.currentAudit) return;
      state.currentAudit.mode = "edit";
      state.currentAudit.step = 2;
      renderActiveTab();
    }

    function renderStep1() {
      const audit = state.currentAudit;
      const data = state.currentAudit.caseData;
      const e = state.currentAudit.errors || {};
      const isText = data.tipoAuditoria === "Escrita Mensageria";
      return `
        <div class="form-section">
          ${auditTypeControl(data, audit.manualCreated, e.tipoAuditoria)}
          ${auditInput("Auditor", "auditorNome", userById(audit.auditorId)?.nome || state.currentUser.nome, true, e.auditorNome)}
          ${isText ? auditInput("Cliente", "cliente", data.cliente || data.atendente, false, e.cliente) : auditInput("Atendente", "atendente", data.atendente, false, e.atendente)}
          ${auditInput("BPO", "bpo", data.bpo, false, e.bpo)}
          ${isText ? auditInput("Proposta/Consulta", "propostaConsulta", data.propostaConsulta, false, e.propostaConsulta, "wide") : auditSelect("Tipo da ligação", "tipoLigacao", data.tipoLigacao, [["Ativa", "Ativa"], ["Receptiva", "Receptiva"]], e.tipoLigacao)}
          ${!isText ? auditInput("Tempo de ligação", "tempoLigacao", data.tempoLigacao, false, e.tempoLigacao) : ""}
          ${isText ? auditInput("Contrato", "contrato", data.contrato, false, e.contrato) : ""}
          ${auditInput("ID gerado", "idCaso", data.idCaso, true, e.idCaso)}
        </div>
        <div class="actions-row">
          <button class="btn btn-ghost" onclick="cancelAudit()">${icon("ph-arrow-left")} Cancelar</button>
          <button class="btn btn-primary" onclick="goChecklist()">${icon("ph-arrow-right")} Continuar</button>
        </div>
      `;
    }

    function auditTypeControl(data, editable, error) {
      if (!editable) return auditInput("Tipo de auditoria", "tipoAuditoria", labels[data.tipoAuditoria] || data.tipoAuditoria, true, error);
      return `
        <div class="form-row ${error ? "invalid" : ""}">
          <label>Tipo de auditoria</label>
          <select onchange="updateAuditType(this.value)">
            ${Object.keys(criteriaCatalog).map(type => `<option value="${type}" ${data.tipoAuditoria === type ? "selected" : ""}>${labels[type]}</option>`).join("")}
          </select>
          <div class="error-msg">${error || ""}</div>
        </div>
      `;
    }

    function auditSelect(label, key, value, options, error, extraClass = "") {
      return `
        <div class="form-row ${extraClass} ${error ? "invalid" : ""}">
          <label>${label}</label>
          <select onchange="updateAuditField('${key}', this.value)">
            ${options.map(([optionValue, optionLabel]) => `<option value="${optionValue}" ${value === optionValue ? "selected" : ""}>${optionLabel}</option>`).join("")}
          </select>
          <div class="error-msg">${error || ""}</div>
        </div>
      `;
    }

    function auditInput(label, key, value, readonly, error, extraClass = "") {
      const dataKey = readonly || key === "tipoAuditoria" || key === "auditorNome" ? "" : `oninput="updateAuditField('${key}', this.value)"`;
      return `
        <div class="form-row ${extraClass} ${error ? "invalid" : ""}">
          <label>${label}</label>
          <input value="${escapeHTML(value)}" ${readonly ? "readonly" : ""} ${dataKey}>
          <div class="error-msg">${error || ""}</div>
        </div>
      `;
    }

    function updateAuditField(key, value) {
      if (!state.currentAudit) return;
      state.currentAudit.caseData[key] = value;
      if (state.currentAudit.errors) state.currentAudit.errors[key] = "";
      markAuditDirty();
    }

    function validateStep1() {
      const data = state.currentAudit.caseData;
      const required = data.tipoAuditoria === "Escrita Mensageria"
        ? ["cliente", "bpo", "tipoAuditoria", "idCaso", "propostaConsulta", "contrato"]
        : ["atendente", "bpo", "tipoAuditoria", "tipoLigacao", "tempoLigacao", "idCaso"];
      const errors = {};
      required.forEach(key => {
        if (!String(state.currentAudit.caseData[key] || "").trim()) errors[key] = "Campo obrigatório.";
      });
      state.currentAudit.errors = errors;
      return Object.keys(errors).length === 0;
    }

    function goChecklist() {
      if (!validateStep1()) {
        renderActiveTab();
        showToast("Preencha os campos obrigatórios para continuar.", "warning");
        return;
      }
      state.currentAudit.step = 2;
      autosaveNow();
      renderActiveTab();
    }

    function renderStep2() {
      const audit = state.currentAudit;
      if (audit.caseData.tipoAuditoria === "Escrita Mensageria") {
        return `
          <div class="check-toolbar">
            <div>
              <div class="caption">Modo de avaliação</div>
              <div class="segmented" style="margin-top:8px">
                <button class="${!audit.manualSimplified ? "active" : ""}" onclick="setManualMode(false)">Modo Completo</button>
                <button class="${audit.manualSimplified ? "active" : ""}" onclick="setManualMode(true)">Modo Simplificado</button>
              </div>
            </div>
            ${autosaveIndicator()}
          </div>
          ${audit.manualSimplified ? renderSimplifiedStep() : renderChecklistStep()}
        `;
      }
      return `
        <div class="check-toolbar">
          ${renderChecklistProgress()}
          <div style="display:flex; gap:10px; flex-wrap:wrap">
            ${autosaveIndicator()}
            <button class="btn btn-secondary" onclick="confirmMarkAll()">${icon("ph-checks")} Marcar todos como Atendeu</button>
          </div>
        </div>
        ${renderChecklistStep(false)}
      `;
    }

    function renderChecklistStep(showToolbar = true) {
      const audit = state.currentAudit;
      return `
        ${showToolbar ? `
          <div class="check-toolbar">
            ${renderChecklistProgress()}
            <div style="display:flex; gap:10px; flex-wrap:wrap">
              ${autosaveIndicator()}
              <button class="btn btn-secondary" onclick="confirmMarkAll()">${icon("ph-checks")} Marcar todos como Atendeu</button>
            </div>
          </div>
        ` : ""}
        <div class="criterion-list">
          ${audit.criteria.map((item, index) => renderCriterion(item, index)).join("")}
        </div>
        <div class="actions-row">
          <button class="btn btn-secondary" onclick="state.currentAudit.step=1; renderActiveTab();">${icon("ph-arrow-left")} Voltar</button>
          <button class="btn btn-primary" onclick="goResult()">${icon("ph-chart-bar")} Ver Resultado</button>
        </div>
      `;
    }

    function renderSimplifiedStep() {
      const audit = state.currentAudit;
      const e = audit.errors || {};
      return `
        <div class="grid grid-2">
          <div class="form-row ${e.manualScore ? "invalid" : ""}">
            <label>Nota manual (0 a 100)</label>
            <input type="number" min="0" max="100" value="${escapeHTML(audit.manualScore)}" oninput="updateManualScore(this.value)" placeholder="Ex.: 87">
            <div class="error-msg">${e.manualScore || ""}</div>
          </div>
          <div class="summary-tile">
            <span class="caption">Classificação prevista</span>
            <strong style="margin-top:8px">${audit.manualScore !== "" ? classificationBadge(classify(Number(audit.manualScore))) : `<span class="badge neutral">Aguardando nota</span>`}</strong>
          </div>
        </div>
        <div class="form-row ${e.observacaoGeral ? "invalid" : ""}" style="margin-top:18px">
          <label>Observação geral obrigatória</label>
          <textarea oninput="updateGeneralObs(this.value)" placeholder="Descreva a justificativa da nota manual.">${escapeHTML(audit.observacaoGeral)}</textarea>
          <div class="error-msg">${e.observacaoGeral || ""}</div>
        </div>
        <div class="actions-row">
          <button class="btn btn-secondary" onclick="state.currentAudit.step=1; renderActiveTab();">${icon("ph-arrow-left")} Voltar</button>
          <button class="btn btn-primary" onclick="goResult()">${icon("ph-chart-bar")} Ver Resultado</button>
        </div>
      `;
    }

    function renderChecklistProgress() {
      const filled = state.currentAudit.criteria.filter(item => item.resposta).length;
      const total = state.currentAudit.criteria.length;
      const pct = total ? (filled / total) * 100 : 0;
      return `
        <div class="progress-wrap">
          <div class="progress-track"><div class="progress-fill" style="--w:${pct}%"></div></div>
          <div class="progress-caption">
            <span>${filled}/${total} preenchidos</span>
            <span>${Math.round(pct)}%</span>
          </div>
        </div>
      `;
    }

    function autosaveIndicator() {
      const last = state.currentAudit?.lastSavedAt;
      const [variant, iconName, label] = saveStatusMeta();
      return `<span class="badge ${variant}">${icon(iconName)} ${last ? `${label} às ${last}` : label}</span>`;
    }

    function renderCriterion(item, index) {
      const open = !!state.currentAudit.openCriteria[index];
      const active = item.resposta;
      return `
        <section id="criterion-${index}" class="criterion ${active ? "answered" : ""} ${open ? "open" : ""}">
          <div class="criterion-head">
            <button class="criterion-name" onclick="toggleCriterion(${index})" style="background:transparent;color:var(--text);text-align:left;padding:0">
              <span class="badge info">${Number(item.peso).toFixed(item.peso % 1 ? 1 : 0)}%</span>
              <strong>${escapeHTML(item.criterioNome)}</strong>
              ${icon(open ? "ph-caret-up" : "ph-caret-down")}
            </button>
            <div class="criterion-actions">
              ${answerButton(index, ANSWERS.OK, "ok", "Atendeu")}
              ${answerButton(index, ANSWERS.PARTIAL, "partial", "Parcial")}
              ${answerButton(index, ANSWERS.NO, "no", "Não atendeu")}
            </div>
          </div>
          <div class="criterion-obs">
            <div class="form-row">
              <label>Observação individual</label>
              <textarea oninput="updateCriterionObs(${index}, this.value)" placeholder="Adicione uma observação para este critério.">${escapeHTML(item.observacaoIndividual || "")}</textarea>
            </div>
          </div>
        </section>
      `;
    }

    function answerButton(index, value, cls, text) {
      const active = state.currentAudit.criteria[index].resposta === value ? "active" : "";
      return `<button class="answer-btn ${cls} ${active}" onclick="setAnswer(${index}, '${value}')">${text}</button>`;
    }

    function setAnswer(index, value) {
      state.currentAudit.criteria[index].resposta = value;
      state.currentAudit.openCriteria[index] = true;
      markAuditDirty();
      autosaveNow(false);
      renderActiveTab();
      requestAnimationFrame(() => {
        const el = document.getElementById(`criterion-${index}`);
        if (el) {
          el.classList.add("sealed");
          setTimeout(() => el.classList.remove("sealed"), 230);
        }
      });
    }

    function updateCriterionObs(index, value) {
      state.currentAudit.criteria[index].observacaoIndividual = value;
      markAuditDirty();
    }

    function toggleCriterion(index) {
      state.currentAudit.openCriteria[index] = !state.currentAudit.openCriteria[index];
      renderActiveTab();
    }

    function setManualMode(enabled) {
      state.currentAudit.manualSimplified = enabled;
      state.currentAudit.errors = {};
      markAuditDirty();
      autosaveNow(false);
      renderActiveTab();
    }

    function updateManualScore(value) {
      state.currentAudit.manualScore = value;
      if (state.currentAudit.errors) state.currentAudit.errors.manualScore = "";
      markAuditDirty();
    }

    function updateGeneralObs(value) {
      state.currentAudit.observacaoGeral = value;
      if (state.currentAudit.errors) state.currentAudit.errors.observacaoGeral = "";
      markAuditDirty();
    }

    function validateStep2() {
      const audit = state.currentAudit;
      const errors = {};
      if (audit.manualSimplified) {
        const score = Number(audit.manualScore);
        if (audit.manualScore === "" || Number.isNaN(score) || score < 0 || score > 100) {
          errors.manualScore = "Informe uma nota entre 0 e 100.";
        }
        if (!String(audit.observacaoGeral || "").trim()) {
          errors.observacaoGeral = "A observação geral é obrigatória no modo simplificado.";
        }
      } else {
        const missing = audit.criteria.filter(item => !item.resposta).length;
        if (missing) errors.criteria = `${missing} critério(s) ainda sem resposta.`;
      }
      audit.errors = errors;
      return Object.keys(errors).length === 0;
    }

    function goResult() {
      if (!validateStep2()) {
        renderActiveTab();
        showToast(state.currentAudit.errors.criteria || "Revise os campos obrigatórios.", "warning");
        return;
      }
      state.currentAudit.step = 3;
      autosaveNow();
      renderActiveTab();
    }

    function calculateCurrentAudit() {
      const audit = state.currentAudit;
      if (!audit) return null;
      if (audit.manualSimplified) {
        const score = Math.max(0, Math.min(100, Number(audit.manualScore || 0)));
        return {
          notaFinal: Math.round(score * 10) / 10,
          classificacao: classify(score),
          counts: { ok: 0, partial: 0, no: 0 },
          criterios: []
        };
      }
      const notaFinal = calculateScore(audit.criteria);
      return {
        notaFinal,
        classificacao: classify(notaFinal),
        counts: {
          ok: audit.criteria.filter(c => c.resposta === ANSWERS.OK).length,
          partial: audit.criteria.filter(c => c.resposta === ANSWERS.PARTIAL).length,
          no: audit.criteria.filter(c => c.resposta === ANSWERS.NO).length
        },
        criterios: audit.criteria
      };
    }

    function renderStep3() {
      const audit = state.currentAudit;
      const result = calculateCurrentAudit();
      const c = audit.caseData;
      const readOnly = audit.mode === "read";
      return `
        <div class="result-hero">
          <div class="score-box">
            <div>
              <div class="caption">Conformidade</div>
              <div class="score-value">${result.notaFinal.toFixed(1)}%</div>
              ${classificationBadge(result.classificacao)}
            </div>
          </div>
          <div class="card" style="box-shadow:none">
            <div class="card-title">
              <h2>Resumo da auditoria</h2>
            </div>
            <div class="summary-grid">
              <div class="summary-tile"><strong>${result.counts.ok}</strong><span class="caption">Atendeu</span></div>
              <div class="summary-tile"><strong>${result.counts.partial}</strong><span class="caption">Parcial</span></div>
              <div class="summary-tile"><strong>${result.counts.no}</strong><span class="caption">Não atendeu</span></div>
            </div>
          </div>
        </div>

        <div class="detail-grid" style="margin-bottom:24px">
          ${detail("ID", c.idCaso)}
          ${detail("Tipo", labels[c.tipoAuditoria] || c.tipoAuditoria)}
          ${detail("BPO", c.bpo)}
          ${c.tipoAuditoria === "Escrita Mensageria" ? detail("Cliente", caseSubject(c)) : detail("Atendente", caseSubject(c))}
          ${c.tipoAuditoria === "Escrita Mensageria" ? detail("Proposta/Consulta", c.propostaConsulta) : detail("Tipo ligação", c.tipoLigacao)}
          ${c.tipoAuditoria === "Escrita Mensageria" ? "" : detail("Tempo ligação", c.tempoLigacao)}
          ${c.tipoAuditoria === "Escrita Mensageria" ? detail("Contrato", c.contrato) : detail("Auditor", userById(audit.auditorId)?.nome || "-")}
        </div>

        <div class="form-row">
          <label>Observação geral</label>
          <textarea ${readOnly ? "readonly" : ""} oninput="updateGeneralObs(this.value)" placeholder="Registre uma conclusão geral da auditoria.">${escapeHTML(audit.observacaoGeral || "")}</textarea>
        </div>

        <div class="actions-row">
          ${readOnly ? `
            <button class="btn btn-secondary" onclick="cancelAudit()">${icon("ph-arrow-left")} Voltar</button>
            <button class="btn btn-secondary" onclick="exportCurrentAudit()">${icon("ph-download-simple")} Exportar esta auditoria</button>
            <button class="btn btn-primary" onclick="editCurrentAudit()">${icon("ph-pencil-simple")} Editar auditoria</button>
          ` : `
            <button class="btn btn-secondary" onclick="state.currentAudit.step=2; renderActiveTab();">${icon("ph-arrow-left")} Voltar ao checklist</button>
            <button class="btn btn-secondary" onclick="exportCurrentAudit()">${icon("ph-download-simple")} Exportar esta auditoria</button>
            <button class="btn btn-primary" onclick="saveCurrentAudit()" ${state.saveLoading ? "disabled" : ""}>
              ${state.saveLoading ? `<span class="spinner"></span> Salvando` : `${icon("ph-floppy-disk")} Salvar auditoria`}
            </button>
          `}
        </div>
      `;
    }

    function detail(label, value) {
      return `<div class="detail-item"><span>${label}</span><strong>${escapeHTML(value)}</strong></div>`;
    }

    function classificationBadge(classificacao) {
      return `<span class="badge ${classVariant(classificacao)}">${icon(classificacao === "Excelente" ? "ph-check-circle" : classificacao === "Regular" ? "ph-warning-circle" : "ph-x-circle")} ${classificacao}</span>`;
    }

    function confirmMarkAll() {
      showModal({
        title: `${icon("ph-checks")} Marcar todos como Atendeu`,
        body: "Esta ação preencherá todos os critérios do checklist atual como Atendeu. Você ainda poderá revisar e alterar qualquer resposta antes de salvar.",
        actions: `
          <button class="btn btn-secondary" onclick="closeModal()">Cancelar</button>
          <button class="btn btn-primary" onclick="markAllOk(); closeModal();">${icon("ph-check-circle")} Confirmar</button>
        `
      });
    }

    function markAllOk() {
      state.currentAudit.criteria.forEach(item => item.resposta = ANSWERS.OK);
      markAuditDirty();
      autosaveNow();
      renderActiveTab();
      showToast("Checklist preenchido como Atendeu.", "success");
    }

    function autosaveNow(show = true) {
      if (!state.currentAudit) return;
      try {
        state.saveStatus = "salvando";
        const saved = DataService.saveDraft(currentDraftPayload());
        const d = new Date(saved.savedAt);
        state.currentAudit.lastSavedAt = d.toLocaleTimeString("pt-BR", { hour: "2-digit", minute: "2-digit" });
        state.currentAudit.dirty = false;
        state.saveStatus = saved.officialSaved ? "salvo" : "aguardando banco de dados";
        if (show) showToast(saved.officialSaved ? "Rascunho salvo." : "Rascunho mantido temporariamente nesta sessão.", "info");
      } catch (err) {
        state.saveStatus = DataService.remoteConnected ? "erro ao salvar" : "sem conexão com o serviço";
        if (show) showToast("Não foi possível salvar o rascunho. Verifique a conexão com o serviço.", "error");
      }
    }

    setInterval(() => {
      if (state.currentUser && state.currentAudit && state.currentAudit.mode !== "read") {
        autosaveNow(false);
        if (state.currentTab === "newAudit" || state.currentTab === "createAudit") renderActiveTab();
      }
    }, 30000);

    window.addEventListener("beforeunload", event => {
      if (!hasUnsavedAudit()) return;
      event.preventDefault();
      event.returnValue = "";
    });

    function saveCurrentAudit() {
      if (!state.currentAudit || state.saveLoading) return;
      const audit = state.currentAudit;
      if (!validateStep2()) {
        audit.step = 2;
        renderActiveTab();
        showToast("Revise a auditoria antes de salvar.", "warning");
        return;
      }
      state.saveLoading = true;
      state.saveStatus = "salvando";
      renderActiveTab();
      setTimeout(() => {
        try {
          const result = calculateCurrentAudit();
          const normalizedCase = { ...audit.caseData };
          if (normalizedCase.tipoAuditoria === "Escrita Mensageria") normalizedCase.atendente = normalizedCase.cliente || normalizedCase.atendente;
          const payload = {
            id: audit.editingAuditId,
            caseId: audit.caseId,
            auditorId: audit.auditorId,
            criterios: audit.manualSimplified ? [] : audit.criteria.map(item => ({ ...item })),
            notaFinal: result.notaFinal,
            classificacao: result.classificacao,
            observacaoGeral: audit.observacaoGeral || "",
            manualSimplified: audit.manualSimplified
          };
          const saved = DataService.concludeAudit(payload, normalizedCase);
          if (state.currentAudit) state.currentAudit.dirty = false;
          refreshStateFromService();
          state.saveStatus = DataService.remoteConnected ? "salvo" : "aguardando banco de dados";
          state.saveLoading = false;
          showModal({
            title: `${icon("ph-check-circle")} Auditoria salva`,
            body: `O caso ${escapeHTML(saved.case.idCaso)} foi concluído. ${DataService.remoteConnected ? "Dados sincronizados com o serviço." : "No protótipo local, os dados permanecem temporários até a conexão com o SQLite remoto."}`,
            actions: `<button class="btn btn-primary" onclick="closeModal()">${icon("ph-arrow-right")} Continuar</button>`,
            onClose: () => {
              state.currentAudit = null;
              state.currentTab = hasManagerAccess() ? "myAudits" : "queue";
              renderApp();
            }
          });
        } catch (err) {
          state.saveLoading = false;
          state.saveStatus = "erro ao salvar";
          renderActiveTab();
          showToast(err.message || "Não foi possível salvar. Verifique a conexão com o serviço.", "error");
        }
      }, 1000);
    }

    function cancelAudit() {
      if (hasUnsavedAudit()) {
        showModal({
          title: `${icon("ph-warning-circle")} Descartar auditoria em andamento?`,
          body: "Existem alterações não salvas oficialmente. Se descartar, a reserva do caso será liberada nesta sessão.",
          actions: `
            <button class="btn btn-secondary" onclick="closeModal()">Continuar editando</button>
            <button class="btn btn-danger" onclick="discardAuditChanges()">${icon("ph-trash")} Descartar</button>
          `
        });
        return;
      }
      clearCurrentAudit({ releaseLock: true });
      state.currentTab = hasManagerAccess() ? "createAudit" : "queue";
      renderApp();
    }

    function clearCurrentAudit({ releaseLock = false } = {}) {
      if (releaseLock && state.currentAudit?.caseId && state.currentUser) {
        DataService.releaseCase(state.currentAudit.caseId, state.currentUser.id);
        refreshStateFromService();
      }
      state.currentAudit = null;
      state.saveLoading = false;
      state.saveStatus = DataService.remoteConnected ? "salvo" : "aguardando banco de dados";
    }

    function discardAuditChanges(targetTab = "") {
      closeModal();
      clearCurrentAudit({ releaseLock: true });
      state.currentTab = targetTab || (hasManagerAccess() ? "createAudit" : "queue");
      renderApp();
    }

    function periodLabel() {
      const period = state.filters.dashboardPeriod;
      if (period === "semana") return "Últimos 7 dias";
      if (period === "mes") return "Últimos 30 dias";
      if (period === "trimestre") return "Últimos 90 dias";
      if (period === "personalizado") {
        const start = state.filters.dashboardStart ? fmtDate(state.filters.dashboardStart) : "início";
        const end = state.filters.dashboardEnd ? fmtDate(state.filters.dashboardEnd) : "hoje";
        return `${start} a ${end}`;

    function failingCriteriaSummary(audits) {
      const map = {};
      audits.forEach(a => {
        (a.criterios || []).forEach(item => {
          if (item.resposta === ANSWERS.OK) return;
          const key = item.criterioNome;
          map[key] = map[key] || { name: key, count: 0, total: 0 };
          map[key].count++;
        });
        (a.criterios || []).forEach(item => {
          const key = item.criterioNome;
          map[key] = map[key] || { name: key, count: 0, total: 0 };
          map[key].total++;
        });
      });
      return Object.values(map)
        .filter(row => row.count > 0)
        .map(row => ({ ...row, rate: row.total ? (row.count / row.total) * 100 : 0 }))
        .sort((a, b) => b.count - a.count || b.rate - a.rate);
    }

    function positiveCriteriaSummary(audits) {
      const map = {};
      audits.forEach(a => {
        (a.criterios || []).forEach(item => {
          const key = item.criterioNome;
          map[key] = map[key] || { name: key, ok: 0, total: 0 };
          map[key].total++;
          if (item.resposta === ANSWERS.OK) map[key].ok++;
        });
      });
      return Object.values(map)
        .filter(row => row.total > 0)
        .map(row => ({ ...row, rate: (row.ok / row.total) * 100 }))
        .filter(row => row.rate >= 70)
        .sort((a, b) => b.rate - a.rate || b.ok - a.ok);
    }

    function dashboardCaseMatches(c) {
      const byBpo = state.filters.dashboardBpo === "todos" || c.bpo === state.filters.dashboardBpo;
      const byType = state.filters.dashboardType === "todos" || c.tipoAuditoria === state.filters.dashboardType;
      const byAuditor = state.filters.dashboardAuditor === "todos" || String(c.auditorId) === state.filters.dashboardAuditor;
      const byPeriod = matchesPeriod(c.importadoEm || c.assignedAt, state.filters.dashboardPeriod, state.filters.dashboardStart, state.filters.dashboardEnd);
      return byBpo && byType && byAuditor && byPeriod;
    }

    function dashboardAuditMatches(a) {
      const c = caseById(a.caseId);
      const byCase = dashboardCaseMatches(c);
      const byAuditor = state.filters.dashboardAuditor === "todos" || String(a.auditorId) === state.filters.dashboardAuditor;
      const byPeriod = matchesPeriod(a.concluidaEm, state.filters.dashboardPeriod, state.filters.dashboardStart, state.filters.dashboardEnd);
      return byCase && byAuditor && byPeriod;
    }

    function matchesPeriod(iso, period, start, end) {
      if (!iso || period === "todos") return true;
      if (period === "personalizado") {
        const afterStart = !start || iso >= start;
        const beforeEnd = !end || iso <= end;
        return afterStart && beforeEnd;
      }
      const days = daysBetween(iso);
      if (period === "semana") return days <= 7;
      if (period === "mes") return days <= 30;
      if (period === "trimestre") return days <= 90;
      return true;
    }

    function renderEvolutionChart(audits) {

          </g>

          ${points.map((p, i) => {

    function weeklyAverages(audits) {
      const grouped = {};
      audits.forEach(a => {
        const d = new Date(`${a.concluidaEm}T12:00:00`);
        const week = Math.ceil(d.getDate() / 7);
        const key = `S${week}`;
        grouped[key] = grouped[key] || [];
        grouped[key].push(a.notaFinal);
      });
      return Object.entries(grouped).sort().map(([label, values]) => ({
        label,
        avg: values.reduce((s, v) => s + v, 0) / values.length
      }));
    }

    function handleImportFile(event) {
      const file = event.target.files[0];
      if (!file) return;
      const xlsx = window.XLSX || (typeof XLSX !== "undefined" ? XLSX : null);
      if (!xlsx) return showToast("SheetJS não carregou. Verifique a conexão e tente novamente.", "error");
      const reader = new FileReader();
      reader.onload = e => {
        try {
          const wb = xlsx.read(new Uint8Array(e.target.result), { type: "array" });
          const first = wb.Sheets[wb.SheetNames[0]];
          const rows = xlsx.utils.sheet_to_json(first, { defval: "" });
          if (!rows.length) {
            state.importPreview = [];
            state.importSummary = { valid: 0, invalid: 0, duplicates: 0, missing: [] };
            renderActiveTab();
            showToast("A planilha não possui linhas para importar.", "warning");
            return;
          }
          const headers = Object.keys(rows[0] || {}).map(h => normalizeColumn(h));
          const missing = IMPORT_BASE_COLUMNS.filter(col => !headers.includes(col));
          if (missing.length) {
            state.importPreview = [];
            state.importSummary = { valid: 0, invalid: rows.length, duplicates: 0, missing };
            renderActiveTab();
            showToast(`Planilha incompleta. Campos faltando: ${missing.join(", ")}.`, "error");
            return;
          }
          state.importPreview = validateImportRows(rows);
          state.importSummary = {
            valid: state.importPreview.filter(r => r.valid && !r.duplicate).length,
            invalid: state.importPreview.filter(r => !r.valid).length,
            duplicates: state.importPreview.filter(r => r.duplicate).length,
            missing: []
          };
          renderActiveTab();
          showToast("Planilha validada. Revise a prévia antes de confirmar.", "success");
        } catch (err) {
          showToast("Não foi possível ler o arquivo Excel.", "error");
        }
      };
      reader.readAsArrayBuffer(file);
    }

    function normalizeColumn(name) {
      return String(name || "").trim().toLowerCase().normalize("NFD").replace(/[\u0300-\u036f]/g, "").replace(/\s+/g, "_");
    }

    function getRowValue(row, column) {
      const found = Object.keys(row).find(key => normalizeColumn(key) === column);
      return found ? String(row[found] ?? "").trim() : "";
    }

    function normalizeAuditType(value) {
      const normalized = normalizeColumn(value);
      const map = {
        "telefonica_basica": "Telefonica Basica",
        "telefonica_tecnica": "Telefonica Tecnica",
        "escrita_mensageria": "Escrita Mensageria",
        "escrita": "Escrita Mensageria",
        "mensageria": "Escrita Mensageria",
        "escrita_/_mensageria": "Escrita Mensageria"
      };
      return map[normalized] || "";
    }

    function normalizeCallType(value) {
      const normalized = normalizeColumn(value);
      if (["ativa", "ativo"].includes(normalized)) return "Ativa";
      if (["receptiva", "receptivo"].includes(normalized)) return "Receptiva";
      return String(value || "").trim();
    }

    function duplicateKeyFor(row) {
      if (row.contrato) return `contrato:${row.contrato.toLowerCase()}`;
      if (row.protocolo) return `protocolo:${row.protocolo.toLowerCase()}`;
      if (row.idExterno) return `externo:${row.idExterno.toLowerCase()}`;
      return `combo:${[row.bpo, row.atendente || row.cliente, row.propostaConsulta].join("|").toLowerCase()}`;
    }

    function validateImportRows(rows) {
      const existingKeys = new Set(state.cases.map(duplicateKeyFor));
      const seen = new Set();
      return rows.map((raw, index) => {
        const rawType = getRowValue(raw, "tipo_auditoria");
        const tipoAuditoria = normalizeAuditType(rawType);
        const auditorLogin = getRowValue(raw, "auditor_login");
        const auditor = auditorLogin ? state.users.find(u => u.login === auditorLogin && u.ativo && u.perfil === "auditor") : null;
        const row = {
          rowNumber: index + 2,
          atendente: getRowValue(raw, "atendente"),
          cliente: getRowValue(raw, "cliente"),
          bpo: getRowValue(raw, "bpo"),
          tipoAuditoria: tipoAuditoria || rawType,
          tipoLigacao: normalizeCallType(getRowValue(raw, "tipo_ligacao")),
          tempoLigacao: getRowValue(raw, "tempo_ligacao"),
          propostaConsulta: getRowValue(raw, "proposta_consulta"),
          contrato: getRowValue(raw, "contrato"),
          protocolo: getRowValue(raw, "protocolo"),
          idExterno: getRowValue(raw, "id_externo"),
          auditorLogin,
          auditorId: auditor?.id || null,
          errors: []
        };
        const required = tipoAuditoria ? [...IMPORT_BASE_COLUMNS, ...(IMPORT_TYPE_REQUIREMENTS[tipoAuditoria] || [])] : IMPORT_BASE_COLUMNS;
        required.forEach(col => {
          const value = getRowValue(raw, col);
          if (!value) row.errors.push(`Campo obrigatório ausente: ${col}`);
        });
        if (!tipoAuditoria) row.errors.push("Tipo de auditoria inválido. Use Telefonica Basica, Telefonica Tecnica ou Escrita Mensageria.");
        if (tipoAuditoria !== "Escrita Mensageria" && row.tipoLigacao && !["Ativa", "Receptiva"].includes(row.tipoLigacao)) {
          row.errors.push("Tipo de ligação inválido. Use Ativa ou Receptiva.");
        }
        if (auditorLogin && !auditor) row.errors.push("Auditor informado não existe, está inativo ou não tem perfil auditor.");
        const key = duplicateKeyFor(row);
        row.duplicate = existingKeys.has(key) || seen.has(key);
        if (row.duplicate) row.errors.push("Possível duplicidade detectada.");
        seen.add(key);
        row.valid = row.errors.length === 0;
        return row;
      });
    }

    function confirmImport() {
      const valid = state.importPreview.filter(r => r.valid && !r.duplicate);
      if (!valid.length) return showToast("Nenhum caso válido para importar.", "warning");
      showModal({
        title: `${icon("ph-upload-simple")} Confirmar importação`,
        body: `${valid.length} linha(s) válida(s) serão importadas. Linhas com erro ou duplicidade serão ignoradas.`,
        actions: `
          <button class="btn btn-secondary" onclick="closeModal()">Voltar</button>
          <button class="btn btn-primary" onclick="executeImport()">${icon("ph-check-circle")} Confirmar importação</button>
        `
      });
    }

    function executeImport() {
      const valid = state.importPreview.filter(r => r.valid && !r.duplicate);
      DataService.importCases(valid);
      const count = valid.length;
      state.importPreview = [];
      state.importSummary = null;
      refreshStateFromService();
      closeModal();
      state.currentTab = hasManagerAccess() ? "importCases" : "queue";
      renderApp();
      showToast(`${count} caso(s) importado(s) e distribuído(s).`, "success");
    }

    function exportAudits(audits, filename) {
      try {
        const xlsx = window.XLSX || (typeof XLSX !== "undefined" ? XLSX : null);
        const allCriteriaNames = [...new Set(audits.flatMap(a => (a.criterios || []).map(item => item.criterioNome)))];
        const resumo = audits.map(a => {
          const c = auditCaseData(a);
          const u = userById(a.auditorId);
          const sla = slaInfo(c);
          const row = {
            "ID caso": c.idCaso,
            "Auditor": u?.nome || "-",
            "Login auditor": u?.login || "-",
            "Perfil auditor": profileLabel(u?.perfil || "auditor"),
            "BPO": c.bpo,
            "Atendente": c.atendente || "",
            "Cliente": c.cliente || "",
            "Tipo": labels[c.tipoAuditoria] || c.tipoAuditoria,
            "Tipo ligação": c.tipoLigacao || "",
            "Tempo ligação": c.tempoLigacao || "",
            "Proposta/Consulta": c.propostaConsulta || "",
            "Contrato": c.contrato || "",
            "Status caso": c.status || "",
            "Criado manualmente": c.createdManually ? "Sim" : "Não",
            "Data atribuição": fmtDate(c.assignedAt),
            "SLA": sla.label,
            "Nota": a.notaFinal,
            "Classificação": a.classificacao,
            "Observação geral": a.observacaoGeral || "",
            "Modo simplificado": a.manualSimplified ? "Sim" : "Não",
            "Data conclusão": fmtDate(a.concluidaEm)
          };
          allCriteriaNames.forEach((name, index) => {
            const detail = (a.criterios || []).find(item => item.criterioNome === name);
            const prefix = `Critério ${String(index + 1).padStart(2, "0")}`;
            row[`${prefix} - nome`] = name;
            row[`${prefix} - peso`] = detail?.peso ?? "";
            row[`${prefix} - resposta`] = detail?.resposta || (a.manualSimplified ? "Modo Simplificado" : "");
            row[`${prefix} - observação`] = detail?.observacaoIndividual || "";
          });
          return row;
        });
        const detalhes = audits.flatMap(a => {
          const c = auditCaseData(a);
          const u = userById(a.auditorId);
          if (a.manualSimplified) {
            return [{
              "ID caso": c.idCaso,
              "Auditor": u?.nome || "-",
              "BPO": c.bpo,
              "Atendente/Cliente": caseSubject(c),
              "Tipo": labels[c.tipoAuditoria] || c.tipoAuditoria,
              "Critério": "Modo Simplificado",
              "Peso": 100,
              "Resposta": `Nota manual ${a.notaFinal}`,
              "Observação individual": a.observacaoGeral || "",
              "Classificação": a.classificacao,
              "Data": fmtDate(a.concluidaEm)
            }];
          }
          return a.criterios.map(item => ({
            "ID caso": c.idCaso,
            "Auditor": u?.nome || "-",
            "BPO": c.bpo,
            "Atendente/Cliente": caseSubject(c),
            "Tipo": labels[c.tipoAuditoria] || c.tipoAuditoria,
            "Critério": item.criterioNome,
            "Peso": item.peso,
            "Resposta": item.resposta,
            "Observação individual": item.observacaoIndividual || "",
            "Classificação": a.classificacao,
            "Data": fmtDate(a.concluidaEm)
          }));
        });
        const controle = audits.map(a => {
          const c = auditCaseData(a);
          const criterios = a.criterios || [];
          return {
            "ID caso": c.idCaso,
            "BPO": c.bpo,
            "Atendente/Cliente": caseSubject(c),
            "Atendeu": criterios.filter(item => item.resposta === ANSWERS.OK).length,
            "Parcial": criterios.filter(item => item.resposta === ANSWERS.PARTIAL).length,
            "Não atendeu": criterios.filter(item => item.resposta === ANSWERS.NO).length,
            "Total critérios": criterios.length || (a.manualSimplified ? "Manual" : 0),
            "Nota": a.notaFinal,
            "Classificação": a.classificacao,
            "Observação geral": a.observacaoGeral || ""
          };
        });
        if (!xlsx) {
          downloadHtmlExcel({ Resumo: resumo, Criterios: detalhes, Controle: controle }, filename.replace(/\.xlsx$/i, ".xls"));
          showToast("Arquivo Excel gerado em modo compatibilidade.", "success");
          return;
        }
        const wb = xlsx.utils.book_new();
        const wsResumo = xlsx.utils.json_to_sheet(resumo);
        const wsDetalhes = xlsx.utils.json_to_sheet(detalhes);
        const wsControle = xlsx.utils.json_to_sheet(controle);
        styleSheet(wsResumo, xlsx);
        styleSheet(wsDetalhes, xlsx);
        styleSheet(wsControle, xlsx);
        xlsx.utils.book_append_sheet(wb, wsResumo, "Resumo");
        xlsx.utils.book_append_sheet(wb, wsDetalhes, "Critérios");
        xlsx.utils.book_append_sheet(wb, wsControle, "Controle");
        xlsx.writeFile(wb, filename);
        showToast("Arquivo Excel gerado.", "success");
      } catch (err) {
        showToast("Não foi possível exportar o Excel.", "error");
      }
    }

    function auditCaseData(a) {
      return caseById(a.caseId) || a.caseData || {};
    }

    function downloadHtmlExcel(sheets, filename) {
      const html = `
        <html><head><meta charset="utf-8"></head><body>
          ${Object.entries(sheets).map(([name, rows]) => `
            <h2>${escapeHTML(name)}</h2>
            ${htmlTable(rows)}
          `).join("<br>")}
        </body></html>
      `;
      const blob = new Blob([html], { type: "application/vnd.ms-excel;charset=utf-8" });
      const url = URL.createObjectURL(blob);
      const a = document.createElement("a");
      a.href = url;
      a.download = filename;
      a.click();
      URL.revokeObjectURL(url);
    }

    function htmlTable(rows) {
      if (!rows.length) return "<p>Sem dados</p>";
      const headers = Object.keys(rows[0]);
      return `
        <table border="1" cellspacing="0" cellpadding="4">
          <thead><tr>${headers.map(h => `<th>${escapeHTML(h)}</th>`).join("")}</tr></thead>
          <tbody>
            ${rows.map(row => `<tr>${headers.map(h => `<td>${escapeHTML(row[h] ?? "")}</td>`).join("")}</tr>`).join("")}
          </tbody>
        </table>
      `;
    }

    function styleSheet(ws, xlsxLib = window.XLSX || (typeof XLSX !== "undefined" ? XLSX : null)) {
      if (!ws["!ref"]) return;
      if (!xlsxLib) return;
      const range = xlsxLib.utils.decode_range(ws["!ref"]);
      for (let c = range.s.c; c <= range.e.c; c++) {
        const cell = ws[xlsxLib.utils.encode_cell({ r: 0, c })];
        if (cell) cell.s = { font: { bold: true }, fill: { fgColor: { rgb: "DBEAFE" } } };
      }
      ws["!cols"] = Array.from({ length: range.e.c + 1 }, () => ({ wch: 22 }));
    }

    function showToast(message, type = "info") {
      const host = document.getElementById("toastHost");
      const id = `toast-${Date.now()}-${Math.random().toString(16).slice(2)}`;
      const iconName = type === "success" ? "ph-check-circle" : type === "error" ? "ph-x-circle" : type === "warning" ? "ph-warning-circle" : "ph-info";
      host.insertAdjacentHTML("beforeend", `<div id="${id}" class="toast ${type}">${icon(iconName)} <div>${escapeHTML(message)}</div></div>`);
      setTimeout(() => {
        const el = document.getElementById(id);
        if (!el) return;
        el.classList.add("out");
        setTimeout(() => el.remove(), 230);
      }, 3200);
    }

    function showModal({ title, body, actions, onClose = null }) {

    }

    function closeModal() {
      const host = document.getElementById("modalHost");
      const backdrop = host.querySelector(".modal-backdrop");
      if (backdrop) backdrop.classList.remove("show");
      setTimeout(() => {
        host.innerHTML = "";
        const cb = state.modalOnClose;
        state.modalOnClose = null;
        if (typeof cb === "function") cb();
      }, 190);
    }
